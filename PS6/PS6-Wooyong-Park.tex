\subsection{Part (d)}
I calculate the homoskedastic and cluster-robust standard errors by bootstrapping.
In the case of homoskedastic OLS, I used the parametric bootstrap method, since it directly assumes the homoskedasticity of the error term.
I followed the following algorithms:

\begin{algorithm}[H]
    \caption{Parametric Bootstrap for OLS Standard Errors}
    \label{alg:parametric_bootstrap}
    \begin{algorithmic}[1]
        \State Estimate the initial OLS model $lwage_i = \beta_0 + \beta_1 educ_i + \epsilon_i$ to obtain coefficients $\hat{\beta}_0$, $\hat{\beta}_1$ and residuals $\{\hat{\epsilon}_i\}_{i=1}^n$.
        \For{$b = 1$ to $B$}
            \State Create a bootstrap sample of residuals $\{\epsilon_i^{*b}\}_{i=1}^n$ by sampling with replacement from the original residuals $\{\hat{\epsilon}_i\}_{i=1}^n$.
            \State Generate a new outcome variable $lwage_i^{*b} = \hat{\beta}_0 + \hat{\beta}_1 educ_i + \epsilon_i^{*b}$ for $i=1, \dots, n$.
            \State Re-estimate the OLS model of $lwage^{*b}$ on $educ$ to get the coefficient estimate $\hat{\beta}_1^{*b}$.
            \State Store $\hat{\beta}_1^{*b}$.
        \EndFor
        \State The bootstrapped standard error is the standard deviation of the stored coefficients: $SE(\hat{\beta}_1) = \sqrt{\frac{1}{B-1}\sum_{b=1}^B (\hat{\beta}_1^{*b} - \bar{\beta}_1^*)^2}$.
    \end{algorithmic}
\end{algorithm}

For cluster-robust standard errors, I use a two-stage cluster bootstrap:
\begin{algorithm}[H]
    \caption{Cluster-Robust Bootstrap (Two-Stage Resampling)}
    \label{alg:cluster_bootstrap}
    \begin{algorithmic}[1]
        \State Identify the $M$ unique clusters (states of birth) in the data.
        \For{$b = 1$ to $B$}
            \State Create a bootstrap sample of $M$ cluster IDs, $\{c_1^{*b}, \dots, c_M^{*b}\}$, by sampling with replacement from the original cluster IDs.
            \State Initialize an empty bootstrap dataset, \texttt{data\_b}.
            \For{each sampled cluster ID $c_j^{*b}$ in the list}
                \State Let the original data for this cluster be \texttt{data\_c}.
                \State Create a bootstrap sample of individuals from this cluster by sampling $|\text{\texttt{data\_c}}|$ rows with replacement from \texttt{data\_c}.
                \State Append these resampled individuals to \texttt{data\_b}.
            \EndFor
            \State Re-estimate the OLS model of $lwage$ on $educ$ using the bootstrap dataset \texttt{data\_b} to get the coefficient estimate $\hat{\beta}_1^{*b}$.
            \State Store $\hat{\beta}_1^{*b}$.
        \EndFor
        \State The cluster-robust bootstrapped standard error is the standard deviation of the stored coefficients: $SE_{cluster}(\hat{\beta}_1) = \sqrt{\frac{1}{B-1}\sum_{b=1}^B (\hat{\beta}_1^{*b} - \bar{\beta}_1^*)^2}$.
    \end{algorithmic}
\end{algorithm}

\section{Problem 2}
Let the model be
\begin{equation}
    Y_{ic} = \alpha + \boldsymbol{\beta}'\mathbf{X}_{ic} + \eta_c + \nu_{ic}
\end{equation}
where the composite error term is 
\begin{equation}
    v_{ic} = \eta_c + \nu_{ic}
\end{equation}
The full variance-covariance matrix of the errors is block-diagonal, where $\mathbf{I}_C$ is the identity matrix of size $C$ and $\boldsymbol{\Omega}_c$ is the covariance matrix for a single cluster $c$ (size $N_c$):
\begin{equation}
    \mathbf{\Omega} = \mathbf{I}_C \otimes \boldsymbol{\Omega}_c
\end{equation}

We can write the covariance matrix for a single cluster $c$ as:
\begin{equation}
    \boldsymbol{\Omega}_c = \text{Var}(\mathbf{v}_c) = \sigma_\nu^2 \mathbf{I}_{N_c} + \sigma_\eta^2 \mathbf{J}_{N_c}
\end{equation}


The GLS estimator can be written as follows:

\begin{align}
    \hat{\boldsymbol{\beta}}_{GLS} &= \arg\min_{\boldsymbol{\beta}} \bigl(\mathbf{Y}-\mathbf{X}\boldsymbol{\beta}\bigr)' \mathbf{\Omega}^{-1} \bigl(\mathbf{Y}-\mathbf{X}\boldsymbol{\beta}\bigr)\\
    &= (\mathbf{X}' \mathbf{\Omega}^{-1} \mathbf{X})^{-1} (\mathbf{X}' \mathbf{\Omega}^{-1} \mathbf{Y})
\end{align}

Then, we can think of the GLS estimator as the OLS estimator of the transformed variables:
\begin{align}
    \mathbf{X}^* &= \mathbf{P} \mathbf{X} = \mathbf{\Omega}^{-1/2} \mathbf{X}
\end{align}

The transformation for observations within cluster $c$ is equivalent to some sort of demeaning process:

\begin{equation}
    \mathbf{X}_{ic}^* = \mathbf{X}_{ic} - \lambda_c \overline{\mathbf{X}}_c
\end{equation}

where $\overline{\mathbf{X}}_c$ is the cluster mean of $\mathbf{X}$, and $\lambda_c$ is the quasi-demeaning factor:
\begin{equation}
    \lambda_c = 1 - \frac{\sigma_\nu}{\sqrt{\sigma_\nu^2 + N_c \sigma_\eta^2}}
\end{equation}

Since the regressor is constant within the cluster, we have:
\begin{equation}
    \mathbf{X}_{ic} = \mathbf{X}_c \quad \text{for all } i
\end{equation}

This implies that the cluster mean of the regressor is simply the regressor itself:
\begin{equation}
    \overline{\mathbf{X}}_c = \frac{1}{N_c} \sum_{i=1}^{N_c} \mathbf{X}_{ic} = \mathbf{X}_c
\end{equation}

Substituting this into the transformation for $\mathbf{X}_{ic}^*$:
\begin{align}
    \mathbf{X}_{ic}^* &= \mathbf{X}_c - \lambda_c \mathbf{X}_c \\
    \mathbf{X}_{ic}^* &= (1 - \lambda_c) \mathbf{X}_c
\end{align}
Since $\mathbf{X}_{ic}^*$ is proportional to $\mathbf{X}_c$, the transformed regressor matrix $\mathbf{X}^{*}$ is proportional to the original regressor matrix $\mathbf{X}$:
\begin{equation}
    \mathbf{X}^{*} = (1 - \lambda) \mathbf{X}
\end{equation}
where $\lambda$ is constant across clusters (since $N_c$ is constant across clusters, $N_c = N/C$).
The GLS estimator is the OLS estimator of the transformed model $\mathbf{Y}^* = \mathbf{X}^* \boldsymbol{\beta} + \mathbf{v}^*$:
\begin{equation}
    \hat{\boldsymbol{\beta}}_{GLS} = (\mathbf{X}^{*'} \mathbf{X}^{*})^{-1} (\mathbf{X}^{*'} \mathbf{Y}^{*})
\end{equation}

Substitute the proportional relationships ($\mathbf{X}^* = (1-\lambda) \mathbf{X}$ and $\mathbf{Y}^* = (1-\lambda) \mathbf{Y}$):
\begin{align}
    \hat{\boldsymbol{\beta}}_{GLS} &= \bigl( ((1-\lambda) \mathbf{X})' ((1-\lambda) \mathbf{X}) \bigr)^{-1} \bigl( ((1-\lambda) \mathbf{X})' ((1-\lambda) \mathbf{Y}) \bigr) \\
    &= \bigl( (1-\lambda)^2 \mathbf{X}' \mathbf{X} \bigr)^{-1} \bigl( (1-\lambda)^2 \mathbf{X}' \mathbf{Y} \bigr) \\
    &= \frac{1}{(1-\lambda)^2} (\mathbf{X}' \mathbf{X})^{-1} (1-\lambda)^2 (\mathbf{X}' \mathbf{Y}) \\
    &= (\mathbf{X}' \mathbf{X})^{-1} (\mathbf{X}' \mathbf{Y}) = \hat{\boldsymbol{\beta}}_{OLS}
\end{align}

The estimators are algebraically identical
\begin{equation}
    \hat{\boldsymbol{\beta}}_{GLS} = \hat{\boldsymbol{\beta}}_{OLS}
\end{equation}
which implies that their variances are also equal:
\begin{equation}
    \text{Var}(\hat{\boldsymbol{\beta}}_{GLS}) = \text{Var}(\hat{\boldsymbol{\beta}}_{OLS})
\end{equation}
